cmake_minimum_required(VERSION 3.16)
project(DeepLearningDemo LANGUAGES CXX)
include(CTest)
option(DEMO_BUILD_MNN "Build MNN vision demos" OFF)
option(DEMO_BUILD_NCNN "Build NCNN vision demos" OFF)
if(DEMO_BUILD_MNN)
  add_subdirectory(mnn_example)
endif()
if(DEMO_BUILD_NCNN)
  add_subdirectory(ncnn_example)
endif()
if(BUILD_TESTING)
  add_executable(common_tests tests/test_common.cpp)
  target_compile_features(common_tests PRIVATE cxx_std_17)
  target_include_directories(common_tests PRIVATE common)
  target_sources(common_tests PRIVATE mnn_example/src/face/database/stream/file_stream.cpp)
  target_include_directories(common_tests PRIVATE mnn_example/src/face/database/stream)
  add_test(NAME common_regressions COMMAND common_tests)
endif()
